// Wild encounter ABI: each ZoneDef is 10 bytes in this order:
// chunkId (u16 LE), tableBase, tableCount, typeMask, lureSlotNibbles[4],
// signed lvlOffset. Each EncTable is ten creature IDs, inclusive floor, ceil.
//
// lureSlotNibbles maps an active lure type (LureZone::lureType & 3) to a table
// offset inside [tableBase, tableBase + tableCount): a nibble of 1..4 selects
// the lure-biased table, while an inert lure (no deposited charge) selects
// table offset 0, the base wild table.
//
// First-beta authored content (data/beta-area.toml [encounters]): the five
// tables in contract order base, spirit(0), water(1), wind(2), earth(3), so
// lure type t selects tableBase + (t + 1) through nibbles 1..4. Every table
// keeps floor 2 / ceil 5, the authored inclusive level bounds (party average 3
// sits inside them, so an authored encounter spawns at level 3).
namespace EncounterFXData {
    // Every walkable cell of the authored area (x8..23, y0..11) is an
    // encounter tile, so each of the four chunks the route crosses carries the
    // same zone: 1 (pocket y2..3), 33 (pocket y4..6 and the (12,7) doorway),
    // 65 (the (12,8) doorway and corridor x12..15,y9) and 66 (corridor
    // x16..20,y9). A chunk with encounter tiles but no zone would clear an
    // active lure on entry (WorldEngine::onChunkChange), so the route's chunks
    // must all be authored. The remaining entries are the inert 255 sentinels;
    // the layout stays at the fixed eight zones. typeMask bit t marks lure
    // group t supported (stored, not read at runtime).
    uint8_t zoneDefs[] = {
        1, 0, 0, 5, 15, 1, 2, 3, 4, 0,
        33, 0, 0, 5, 15, 1, 2, 3, 4, 0,
        65, 0, 0, 5, 15, 1, 2, 3, 4, 0,
        66, 0, 0, 5, 15, 1, 2, 3, 4, 0,
        255, 255, 0, 0, 0, 0, 0, 0, 0, 0,
        255, 255, 0, 0, 0, 0, 0, 0, 0, 0,
        255, 255, 0, 0, 0, 0, 0, 0, 0, 0,
        255, 255, 0, 0, 0, 0, 0, 0, 0, 0
    };

    // Authored tables, contract order, ten slots plus floor/ceil (2/5). Only
    // first-area species are used; the material derivation and the beta-area
    // host test pin the same IDs.
    uint8_t tables[] = {
        0, 3, 6, 9, 10, 12, 13, 14, 0, 6, 2, 5,
        9, 9, 0, 6, 9, 13, 3, 9, 14, 9, 2, 5,
        6, 6, 3, 6, 9, 6, 14, 6, 0, 6, 2, 5,
        0, 13, 0, 13, 9, 0, 13, 6, 0, 13, 2, 5,
        3, 14, 3, 14, 9, 3, 14, 6, 3, 14, 2, 5
    };
}
